# below are command needs to run in the terminal

## for FEMNIST dataset 
Note: when using this dataset, lr is 0.02, other datasets lr is 0.01

### Baseline
python main_fed.py --dataset FEMNIST --model femnistnet --num_users 10 --num_samples 100 --epochs 20 --local_ep 5 2>&1 | tee femnist_run_baseline_$(date +%Y%m%d_%H%M%S).log

### FinP
python main_fed.py --dataset FEMNIST --model femnistnet --num_users 10 --num_samples 100 --epochs 20 --local_ep 5 --opt --col --beta=1 --hessian_eig_max_iter 20 --hessian_trace_max_iter 20 --hessian_tol 5e-3 2>&1 | tee femnist_run_finp_$(date +%Y%m%d_%H%M%S).log
Note: using those hassian config in this femnist dataset to speed up the hessian calculation   --hessian_eig_max_iter 20 --hessian_trace_max_iter 20 --hessian_tol 5e-3 
Note: run ablation --beta = 0.5, 0.75, 1

### DP-femnist
python main_fed.py --dataset FEMNIST --model femnistnet --num_users 10 --num_samples 100 --epochs 20 --local_ep 5 --run_dp_baseline --dp_clip 7.5 --dp_noise=0.5 2>&1 | tee femnist_run_dp_$(date +%Y%m%d_%H%M%S).log
Note: run ablation --dp_noise= 0.75, 1, 2
